    //    After 2 ticks → rearLat > DRIFT_THRESHOLD → drift fires
    //    Resetting to worldLat each tick discards this accumulation.
    //

    // ═══════════════════════════════════════════════════════════
    //  TICK
    // ═══════════════════════════════════════════════════════════
 
    @Override
    public void tick() {
        super.tick();
 
        if (!this.isLocalInstanceAuthoritative()) {
            this.setDeltaMovement(Vec3.ZERO);
            return;
        }
 
        boolean forward = false, backward = false,
                left    = false, right    = false, handbrake = false;
 
        if (this.getFirstPassenger() instanceof Player player) {
            forward   = player.zza  >  0f;
            backward  = player.zza  <  0f;
            left      = player.xxa  >  0f;
            right     = player.xxa  <  0f;
            handbrake = player.isJumping();
 
            if (this.tickCount % 2 == 0) displaySpeed(player);
        } else {
            frontLat      *= 0.70f;
            rearLat       *= 0.70f;
            overYawRate   *= 0.70f;
            throttleSmooth = 0f;
        }
 
        tickPhysics(forward, backward, left, right, handbrake);
 
        this.move(MoverType.SELF, this.getDeltaMovement());
 
        if (this.onGround() && this.getDeltaMovement().y < 0)
            this.setDeltaMovement(this.getDeltaMovement().x, 0.0, this.getDeltaMovement().z);
 
        this.applyEffectsFromBlocks();
 
        if (this.level().isClientSide()) {
            spawnWheelParticles();
            spawnExhaust();
        }
    }
 
    // ═══════════════════════════════════════════════════════════
    //  PHYSICS ENGINE
    // ═══════════════════════════════════════════════════════════
 
    private void tickPhysics(boolean forward, boolean backward,
                              boolean left,    boolean right, boolean handbrake) {
 
        // ── 1. Local velocity ─────────────────────────────────────────────
        double yRad  = Math.toRadians(this.getYRot());
        double sinY  = Math.sin(yRad), cosY = Math.cos(yRad);
        Vec3   vel   = this.getDeltaMovement();
 
        double localFwd  = -vel.x * sinY + vel.z * cosY;
        double localY    =  vel.y;
        float  speed     = (float) Math.abs(localFwd);
        boolean goingFwd = localFwd >  0.01;
        boolean goingRev = localFwd < -0.01;
        boolean stopped  = !goingFwd && !goingRev;
 
        // ── 2. Input flags ────────────────────────────────────────────────
        // THROTTLE: engine producing power.
        //   Forward: W pressed, not yet reversing.
        //   Reverse: S pressed, car already reversing OR stopped.
        //   (Both include the stopped case so the car can start moving.)
        boolean throttleActive = (forward  && !goingRev)
                              || (backward && !goingFwd);
        // BRAKING: opposing current motion.
        boolean brakingActive  = (backward && goingFwd)
                              || (forward  && goingRev);
 
        float steerInput = left ? -1f : right ? 1f : 0f;
 
        this.entityData.set(DATA_THROTTLE,    throttleActive);
        this.entityData.set(DATA_BRAKING,     brakingActive);
        this.entityData.set(DATA_STEER_INPUT, steerInput);
 
        // ── 3. Smooth throttle ────────────────────────────────────────────
        float throttleTarget = throttleActive ? 1.0f : 0.0f;
        float ramp = throttleTarget > throttleSmooth ? THROTTLE_RAMP_ON : THROTTLE_RAMP_OFF;
        throttleSmooth = Mth.clamp(
            throttleSmooth + (throttleTarget - throttleSmooth) * ramp, 0f, 1f);
 
        // ── 4. REAL FORWARD DRIVETRAIN CHAIN ─────────────────────────────
        //
        //  wheelRPS  = speed / TYRE_CIRC * 20
        //  engineRPM = wheelRPS × GEAR_RATIOS[gear] × FINAL_DRIVE × 60
        //
        //  Auto-box shifts at REDLINE_RPM (up) and DOWNSHIFT_RPM (down).
        //  Gear drop at upshift is automatic because the new gear ratio
        //  is smaller → rawRPM drops → audible pitch drop.
 
        if (speed < 0.005f) {
            currentGear = 1;
            clutchTimer = 0;
            engineRPM   = IDLE_RPM + throttleSmooth * 600f;
        } else {
            float wheelRPS = speed / TYRE_CIRC * 20f;
            float rawRPM   = wheelRPS * GEAR_RATIOS[currentGear] * FINAL_DRIVE * 60f;
 
            if (clutchTimer > 0) {
                clutchTimer--;
                float blend = clutchTimer / (float) CLUTCH_TICKS;
                engineRPM = Mth.lerp(blend, rawRPM, engineRPM * 0.72f);
            } else if (goingFwd) {
                // Auto-box only shifts going forward
                if (rawRPM >= REDLINE_RPM && currentGear < MAX_GEAR) {
                    currentGear++;
                    clutchTimer = CLUTCH_TICKS;
                    rawRPM = wheelRPS * GEAR_RATIOS[currentGear] * FINAL_DRIVE * 60f;
                } else if (rawRPM <= DOWNSHIFT_RPM && currentGear > 1) {
                    currentGear--;
                    clutchTimer = CLUTCH_TICKS;
                    rawRPM = wheelRPS * GEAR_RATIOS[currentGear] * FINAL_DRIVE * 60f;
                }
                engineRPM = rawRPM;
            } else if (goingRev) {
                // Reverse: cap RPM at MAX_REVERSE_RPM (2000) so pitch stays low.
                // The full gear-1 ratio would give redline at 22.6 km/h — below max
                // reverse speed — causing pitch=1.40 and the accel loop restarts every
                // ~2.6s, which is very noticeable.
                // At 2000 RPM max, pitch stays ≤ 0.87 and loops every ~4.3s.
                float revFraction = Math.min(1f, speed / MAX_REVERSE_SPEED);
                engineRPM = IDLE_RPM + revFraction * (MAX_REVERSE_RPM - IDLE_RPM);
            }
 
            engineRPM = Math.max(engineRPM, IDLE_RPM + throttleSmooth * 500f);
            engineRPM = Mth.clamp(engineRPM, IDLE_RPM, REDLINE_RPM);
        }
 
        this.entityData.set(DATA_ENGINE_RPM, engineRPM);
        this.entityData.set(DATA_GEAR,       currentGear);
 
        // ── 5. Drive force ────────────────────────────────────────────────
        float torqueNorm = lookupTorque(engineRPM);
        float gearMult   = GEAR_RATIOS[currentGear] / GEAR_RATIOS[1];
        float driveForce = torqueNorm * throttleSmooth * PEAK_DRIVE_FORCE * gearMult;
        if (clutchTimer > 0) driveForce = 0f;
 
        // ── 6. Longitudinal weight transfer ───────────────────────────────
        float wt          = driveForce - (brakingActive ? 0.04f : 0f);
        float rearWeight  = Mth.clamp(REAR_BIAS  + wt * LONG_TRANSFER, 0.30f, 0.70f);
        float frontWeight = Mth.clamp(FRONT_BIAS - wt * LONG_TRANSFER, 0.30f, 0.70f);
 
        // ── 7. Traction circle (RWD) ──────────────────────────────────────
        float rearGripTotal = rearWeight * REAR_GRIP_MAX;
        float driveUsed     = Math.min(Math.abs(driveForce), rearGripTotal);
        float rearGripLat;
        if (handbrake) {
            // Handbrake: near-zero lateral grip → rear slides freely → drift
            rearGripLat = HANDBRAKE_REAR_GRIP * REAR_GRIP_MAX; // ~0.005
        } else {
            // Traction circle: lateral capacity = sqrt(total² − drive²).
            // driveForce is always larger than rearGripTotal so this would
            // always give 0 without the floor. The floor (40% of total) ensures
            // the rear retains enough lateral grip to corner normally under throttle.
            // Handbrake bypasses this so it still gets the ~0 rear grip it needs.
            rearGripLat = (float) Math.sqrt(Math.max(0.0,
                (double) rearGripTotal * rearGripTotal - (double) driveUsed * driveUsed));
            rearGripLat = Math.max(rearGripLat, rearGripTotal * 0.40f);
        }
        float frontGripLat = frontWeight * FRONT_GRIP_MAX;
 
        // Lateral weight transfer: reduces grip slightly in hard cornering
        float latAbs       = Math.max(Math.abs(frontLat), Math.abs(rearLat));
        float latReduction = Mth.clamp(latAbs * LAT_TRANSFER, 0f, 0.18f);
        frontGripLat *= (1f - latReduction);
        rearGripLat  *= (1f - latReduction);
 
        // ── 8. Longitudinal motion ────────────────────────────────────────
        // FIX: reverse uses the same drive force path.
        // throttleActive is true both when going forward AND when pressing S
        // while reversing/stopped, so both directions work identically.
 
        if (throttleActive) {
            if (forward) {
                // Forward thrust (capped by traction at launch)
                double cap = goingRev ? 0.10 : rearGripTotal;
                localFwd = Math.min(localFwd + Math.min(driveForce, cap), 75.0);
            } else {
                // Reverse thrust — S held, car going backward or stopped
                localFwd = Math.max(localFwd - 0.028, -0.35);
            }
        }
 
        if (brakingActive) {
            if (goingFwd)      localFwd = Math.max(0.0, localFwd - 0.10);
            else if (goingRev) localFwd = Math.min(0.0, localFwd + 0.10);
        }
 
        // Engine braking when coasting (proportional to gear ratio)
        if (!forward && !backward && !handbrake) {
            double eb = 0.004 * (GEAR_RATIOS[currentGear] / GEAR_RATIOS[MAX_GEAR]);
            if (goingFwd)      localFwd -= eb;
            else if (goingRev) localFwd += eb;
        }
 
        // Handbrake: applies real deceleration while allowing the rear to slide.
        // Without this, SPACE did nothing to slow the car — letting go of W
        // stopped it faster than holding SPACE because engine braking > 0 drag.
        if (handbrake) {
            if (goingFwd)
                localFwd = Math.max(0.0, localFwd - 0.06);
            else if (goingRev)
                localFwd = Math.min(0.0, localFwd + 0.06);
        }
 
        // Rolling drag + aero — no hardcoded top speed
        localFwd *= ROLLING_DRAG;
        if (Math.abs(localFwd) > AERO_DRAG_START) {
            double excess = Math.abs(localFwd) - AERO_DRAG_START;
            localFwd -= Math.signum(localFwd) * excess * AERO_DRAG_K * 20;
        }
        if (Math.abs(localFwd) < 0.001) localFwd = 0.0;
 
        // ── 9. STEERING — direct yaw, no grip required ────────────────────
        boolean drifting = Math.abs(rearLat) > DRIFT_THRESHOLD;
        if (speed > 0.01f && steerInput != 0f) {
            float steerDeg = computeSteerAngle(speed, drifting);
            float revSign  = goingRev ? -1f : 1f;
            this.setYRot(this.getYRot() + steerInput * steerDeg * revSign);
        }
 
        // ── 10. CENTRIPETAL LATERAL + PERSISTENT AXLE MODEL ──────────────
        //
        //  HOW DRIFT WORKS WITH THIS MODEL:
        //
        //  Every tick the car steers, it generates centripetal lateral force:
        //    centripetalLat = sin(steerRad) × speed × steerSign
        //
        //  This is injected into BOTH axles. Each axle's grip budget corrects
        //  as much as it can. The remainder persists to the next tick (× LAT_DECAY).
        //
        //  Normal cornering:
        //    rearGripLat ≈ 0.12 → corrects ~0.03 per tick → rearLat stays ~0
        //
        //  Handbrake:
        //    rearGripLat ≈ 0.005 → corrects only ~0.003 per tick
        //    → rearLat accumulates ~0.025/tick
        //    → after 2 ticks rearLat > DRIFT_THRESHOLD → drift detected + yaw fires
        //
        //  Power-on oversteer (excess throttle in 1st/2nd):
        //    driveUsed ≈ rearGripTotal → rearGripLat ≈ 0 → same as handbrake
        //
        //  WHY NOT USE WORLD LATERAL:
        //    localLat from world velocity ≈ 0 during normal play because the
        //    correctedLat output already corrected it last tick. The centripetal
        //    injection creates fresh lateral that accumulates independently.
 
        if (goingFwd && steerInput != 0f) {
            float steerRad       = (float) Math.toRadians(computeSteerAngle(speed, drifting));
            float centripetalLat = (float)(Math.sin(steerRad) * Math.abs(localFwd)) * steerInput;
            frontLat += centripetalLat * (REAR_DIST  / (FRONT_DIST + REAR_DIST));
            rearLat  += centripetalLat * (FRONT_DIST / (FRONT_DIST + REAR_DIST));
        }
 
        // Front axle correction — full grip budget
        float frontCorrect = computeGripForce(frontLat, frontGripLat);
        frontLat -= frontCorrect;
 
        // Rear axle correction — grip budget reduced by traction circle / handbrake
        float rearCorrect;
        if (handbrake) {
            rearCorrect = rearLat * HANDBRAKE_REAR_GRIP * GRIP_STIFFNESS;
        } else {
            rearCorrect = computeGripForce(rearLat, rearGripLat);
        }
        rearLat -= rearCorrect;
 
        // Decay remaining lateral (rolling + aero resistance on lateral motion)
        frontLat *= LAT_DECAY;
        rearLat  *= LAT_DECAY;
 
        if (Math.abs(frontLat) < 0.0005f) frontLat = 0f;
        if (Math.abs(rearLat)  < 0.0005f) rearLat  = 0f;
 
        // Net lateral velocity sent to world movement
        float overallLat = frontLat * FRONT_BIAS + rearLat * REAR_BIAS;
 
        // ── 11. OVERSTEER YAW — gated behind DRIFT_THRESHOLD ─────────────
        //
        //  Only fires when |rearLat| > DRIFT_THRESHOLD (genuine traction loss).
        //  Normal cornering: rearLat ≈ 0 → gate stays closed.
        //  Handbrake/oversteer: rearLat builds → gate opens → yaw fires.
        //  Counter-steer: opposite steerInput → steering yaw (step 9) goes
        //  negative → heading corrects → centripetal injection reduces → rearLat
        //  falls back below threshold → drift exits naturally.
 
        drifting = Math.abs(rearLat) > DRIFT_THRESHOLD;
        if (drifting) {
            float excessDiff = rearLat - frontLat;
            overYawRate += excessDiff * YAW_MOMENT_SCALE;
        } else {
            overYawRate *= 0.55f; // snap back when not drifting
        }
        overYawRate *= YAW_DAMPING;
        overYawRate  = Mth.clamp(overYawRate, -YAW_MAX, YAW_MAX);
        this.setYRot(this.getYRot() + overYawRate);
 
        this.entityData.set(DATA_DRIFTING, drifting || Math.abs(overYawRate) > 1.2f);
 
        // ── 12. Gravity ───────────────────────────────────────────────────
        localY = this.onGround() ? GROUND_STICK : Math.max(-MAX_FALL_SPEED, localY - GRAVITY);
 
        // ── 13. Reproject to world ────────────────────────────────────────
        double nYRad = Math.toRadians(this.getYRot());
        double nSin  = Math.sin(nYRad), nCos = Math.cos(nYRad);
        this.setDeltaMovement(
            localFwd * (-nSin) + overallLat * nCos,
            localY,
            localFwd *   nCos  + overallLat * nSin
        );
 
        // ── 14. Cosmetics ─────────────────────────────────────────────────
        wheelSpin = (wheelSpin + speed * 180f) % 360f;
        this.entityData.set(DATA_WHEEL_SPIN,    wheelSpin);
        this.entityData.set(DATA_FORWARD_SPEED, (float) localFwd);
    }
 
    // ═══════════════════════════════════════════════════════════
    //  HELPERS
    // ═══════════════════════════════════════════════════════════
 
    private float computeGripForce(float slip, float gripMax) {
        float raw = Mth.clamp(slip, -gripMax, gripMax);
        if (Math.abs(slip) > SLIP_THRESHOLD) {
            float falloff = SLIP_FALLOFF + (1f - SLIP_FALLOFF) * (SLIP_THRESHOLD / Math.abs(slip));
            raw *= falloff;
        }
        return raw * GRIP_STIFFNESS;
    }
 
    private float lookupTorque(float rpm) {
        if (rpm <= TC_RPM[0]) return TC_TRQ[0];
        for (int i = 1; i < TC_RPM.length; i++) {
            if (rpm <= TC_RPM[i]) {
                float t = (rpm - TC_RPM[i-1]) / (TC_RPM[i] - TC_RPM[i-1]);
                return TC_TRQ[i-1] + t * (TC_TRQ[i] - TC_TRQ[i-1]);
            }
        }
        return TC_TRQ[TC_TRQ.length - 1];
    }
 
    private float computeSteerAngle(float speed, boolean drifting) {
        float steer;
        if (speed <= PEAK_STEER_SPEED) {
            steer = MAX_STEER_DEG * (float)(speed / PEAK_STEER_SPEED);
        } else {
            double t = Mth.clamp(
                (speed - PEAK_STEER_SPEED) / (1.0 - PEAK_STEER_SPEED), 0.0, 1.0);
            steer = MAX_STEER_DEG * (float)(1.0 - t * (1.0 - HIGH_STEER_FRACTION));
        }
        return drifting ? steer * DRIFT_STEER_BOOST : steer;
    }
 
    // ═══════════════════════════════════════════════════════════
    //  PARTICLES
    // ═══════════════════════════════════════════════════════════
 
    // ── Half track width (left/right from centreline to wheel, in blocks) ──
    // Tune this until the flame debug particles sit over the visual wheels.
    private static final double TRACK_HALF = 0.90;
 
    private void spawnWheelParticles() {
        if (!this.onGround()) return;
        double speed = Math.abs(this.getForwardSpeed());
        if (speed < 0.05) return;
        boolean drifting = this.isDrifting();
 
        double yRad = Math.toRadians(this.getYRot());
        double sinY = Math.sin(yRad), cosY = Math.cos(yRad);
 
        // Forward unit vector in world space: (-sinY, 0, cosY)
        // Right   unit vector in world space: ( cosY, 0, sinY)
        //
        // FRONT_DIST = distance from entity centre → front axle (ahead)
        // REAR_DIST  = distance from entity centre → rear  axle (behind)
        //
        // Wheel world positions:
        //   front-left  = centre + FRONT_DIST*fwd + TRACK_HALF*right
        //   front-right = centre + FRONT_DIST*fwd - TRACK_HALF*right
        //   rear-left   = centre - REAR_DIST *fwd + TRACK_HALF*right
        //   rear-right  = centre - REAR_DIST *fwd - TRACK_HALF*right
 
        double fwdX =  -sinY, fwdZ = cosY;
        double rgtX =   cosY, rgtZ = sinY;
 
        // Front axle
        double fax = fwdX * FRONT_DIST, faz = fwdZ * FRONT_DIST;
        // Rear axle
        double rax = -fwdX * REAR_DIST, raz = -fwdZ * REAR_DIST;
        // Lateral offsets
        double lox = rgtX * TRACK_HALF, loz = rgtZ * TRACK_HALF;
 
        spawnWheelAt( fax + lox, 0.05,  faz + loz, drifting); // front-left
        spawnWheelAt( fax - lox, 0.05,  faz - loz, drifting); // front-right
        spawnWheelAt( rax + lox, 0.05,  raz + loz, drifting); // rear-left
        spawnWheelAt( rax - lox, 0.05,  raz - loz, drifting); // rear-right
 
        // ── DEBUG: flame particles at physics wheel positions ─────────
        // Remove this block once the wheels are correctly positioned.
        if (this.tickCount % 3 == 0) {
            spawnDebugFlame(fax + lox, 0.05,  faz + loz); // front-left
            spawnDebugFlame(fax - lox, 0.05,  faz - loz); // front-right
            spawnDebugFlame(rax + lox, 0.05,  raz + loz); // rear-left
            spawnDebugFlame(rax - lox, 0.05,  raz - loz); // rear-right
        }
    }
 
    private void spawnDebugFlame(double ox, double oy, double oz) {
        Vec3 pos = this.position();
        this.level().addParticle(
            ParticleTypes.FLAME,
            pos.x + ox, pos.y + oy + 0.3, pos.z + oz,
            0, 0.05, 0
        );
    }
 
    private void spawnWheelAt(double ox, double oy, double oz, boolean drifting) {
        Vec3 pos = this.position();
        double px = pos.x+ox, py = pos.y+oy, pz = pos.z+oz;
        Vec3 vel = this.getDeltaMovement();
        double vx = -vel.x*0.5 + (random.nextDouble()-0.5)*0.1;
        double vz = -vel.z*0.5 + (random.nextDouble()-0.5)*0.1;
        if (drifting) { vx *= 1.5; vz *= 1.5; }
        BlockState bs = this.level().getBlockState(BlockPos.containing(px, py-0.2, pz));
        if (drifting)
            this.level().addParticle(ParticleTypes.CAMPFIRE_COSY_SMOKE, px, py, pz, vx, 0.1, vz);
        this.level().addParticle(new BlockParticleOption(ParticleTypes.BLOCK, bs),
            px, py, pz, vx, 0.1, vz);
    }
 
    private void spawnExhaust() {
        double speed = Math.abs(this.getForwardSpeed());
        if (speed < 0.02) return;
        double yRad = Math.toRadians(this.getYRot());
        double px = this.getX() - Math.cos(yRad)*1.2;
        double pz = this.getZ() - Math.sin(yRad)*1.2;
        Vec3 vel = this.getDeltaMovement();
        this.level().addParticle(ParticleTypes.SMOKE, px, this.getY()+0.3, pz,
            -vel.x*0.2+(random.nextDouble()-0.5)*0.02, 0.02,
            -vel.z*0.2+(random.nextDouble()-0.5)*0.02);
    }
 
    // ═══════════════════════════════════════════════════════════
    //  PASSENGER
    // ═══════════════════════════════════════════════════════════
 
    @Override
    protected Vec3 getPassengerAttachmentPoint(Entity p, EntityDimensions d, float t) {
        int i = this.getPassengers().indexOf(p);
        double lx = (i == 0) ? SEAT_SIDE : -SEAT_SIDE;
        double yRad = Math.toRadians(this.getYRot());
        double sin = Math.sin(yRad), cos = Math.cos(yRad);
        return new Vec3(lx*cos + SEAT_FORWARD*(-sin), SEAT_HEIGHT,
                        lx*sin + SEAT_FORWARD*   cos);
    }
 
    @Override protected boolean canAddPassenger(Entity p) { return this.getPassengers().size() < 2; }
 
    @Override @Nullable
    public LivingEntity getControllingPassenger() {
        return this.getFirstPassenger() instanceof LivingEntity le ? le : null;
    }

    private void displaySpeed(Player player) {

        float rpm  = this.getEngineRPM();
        int   gear = this.getCurrentGear();
        player.displayClientMessage(
            net.minecraft.network.chat.Component.literal(String.format(
                "§aSpeed: §f%.0f km/h  §7|  §cRPM: §f%.0f  §7|  §eGear: §f%d",
                kmh, rpm, gear)),
            true);
    }
